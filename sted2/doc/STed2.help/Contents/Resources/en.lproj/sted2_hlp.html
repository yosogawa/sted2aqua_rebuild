   [CTRL]+[ B ]/[OPT.1]+[R.DW]  Previous measure.
         [CTRL]+[ C ]/[R.UP]    Page forward.
         [CTRL]+[ R ]/[R.DW]    Page back.

 [SHIFT]+[R.UP]/[OPT.1]+[↓]    Fast scroll up

         [CTRL]+[TAB]           Set a tag


                [F 1]           First line.
                [F 2]           Last line.

                [F 4]           Set the find text

        [SHIFT]+[F 5]           Find previous

                [F 7]           Cut (move into the copy buffer)
                [F 8]           Copy (copy into the copy buffer)
        [SHIFT]+[F 8]           Mix paste (mix the copy buffer into the track)
                [F 9]           Paste (insert the copy buffer)
        [SHIFT]+[F 9]           Reverse paste (insert the copy buffer reversed)
                [F10]           Duplicate the line at the cursor.
        [SHIFT]+[F 1]           Set velocity across a range
        [SHIFT]+[F 2]           Optimize
        [SHIFT]+[F 4]           Sort notes and shift steps
        [SHIFT]+[F 6]           Store undo (save into the undo buffer)
        [SHIFT]+[F 7]           Undo (swap the undo buffer and the track)

                [ S ]           Save part data
                [ L ]           Load part data
                [ O ]           Export the part as text
                [ R ]           Replicate
        [SHIFT]+[ R ]           Replicate every track

                [ Q ]           Quantize
                [ \ ]           Split pitch bend / control change
        [SHIFT]+[ \ ]           Split a note
                [ / ]           Write a special controller on the cursor line.

        [OPT.1]+[ A ]〜[ Z ]    Run an option command directly
         [CTRL]+[ ] ]           Find the matching repeat

                [ [ ]           Insert a repeat start on the cursor line.
                [ ] ]           Insert a repeat end on the cursor line.
                [ * ]           Draw a bar line on the cursor line.
                [ ^ ]           Set a same-measure copy on the cursor line.
        [SHIFT]+[ ^ ]           Convert duplicates to same-measure copies.
         [CTRL]+[ ^ ]           Expand same-measure copies.
                [ + ]/[ > ]     Add 1 to the value at the cursor.
                [ - ]/[ < ]     Subtract 1 from the value at the cursor.
        [SHIFT]+[ > ]           Add 10 to the value at the cursor.
        [SHIFT]+[ < ]           Subtract 10 from the value at the cursor.
                [ = ]           Flip the sign at the cursor.
                [ 0 ]〜[ 9 ]    Edit the number at the cursor.
                [ A ]〜[ G ]    Edit the note on this line by letter.
                [RET]           Insert a note on the cursor line.

        《Rhythm track editor》

                [←]            Cursor left.
                [→]            Cursor right.
                [↑]            Cursor up.
                [↓]            Cursor down.

                [HOME]          Move the cursor home.
        [SHIFT]+[HOME]          Jump to the play position
                [かな/KANA]      Follow the play position

                [TAB]           Go to a measure number.
         [CTRL]+[ C ]/[R.UP]    Show the next measure's pattern.
         [CTRL]+[ R ]/[R.DW]    Show the previous measure's pattern.
        [SHIFT]+[R.UP]          Show the other half of the pattern.
        [SHIFT]+[R.DW]          Show the other half of the pattern.

                [登録/TOUROKU]   Create a pattern on a new measure.
        [SHIFT]+[登録/TOUROKU]   Create a pattern on a new measure. (Change ST)

                [ESC]           Back to the track editor.

                [ + ]/[ > ]     Add 1 to the value at the cursor.
                [ - ]/[ < ]     Subtract 1 from the value at the cursor.
        [SHIFT]+[ > ]           Add 10 to the value at the cursor.
        [SHIFT]+[ < ]           Subtract 10 from the value at the cursor.

         [CTRL]+[ Y ]/[DEL]     Move the cursor line into the delete-line buffer
         [CTRL]+[ L ]/[UNDO]    Insert the delete-line buffer on the cursor line.

                [F 1]           Show the first measure's pattern.
                [F 2]           Show the last measure's pattern.
                [F 7]           Cut the pattern (move into the copy buffer)
                [F 8]           Copy (copy into the copy buffer)
                [F 9]           Paste (insert the copy buffer)
                [F10]           Duplicate the line at the cursor.

        [SHIFT]+[ / ]           Open rhythm assign.

                [SPACE]         Change the velocity level. (Cycles -,1,2 to 8,9....)
                [ 0 ]〜[ 9 ]    Set the velocity level.
                [RET]           Enter velocity directly, or change the step time.

                [XF1]/[ M ]     Measure play
        [SHIFT]+[XF1]           Point play (this track)
         [CTRL]+[XF1]           Point play (this track, from a measure)
                [XF2]/[ P ]     Point play (all tracks)
        [SHIFT]+[XF2]           Point play (from a measure)
                [XF3]           Continue
                [XF4]           Pause
                [XF5]           Stop

                        ┌───────┬──────────┐
                        │              │       Length       │
                        │     Note     ├──────┬───┤
                        │              │  Fraction  │  ST  │
                        ├───────┼──────┼───┤
                        │  whole note  │     1      │  192 │
                        │ dotted half  │ 1/2 + 1/4  │  144 │
                        │  half note   │    1/2     │   96 │
                        │dotted quarter│ 1/4 + 1/8  │   72 │
                        │ quarter note │    1/4     │   48 │
                        │dotted eighth │ 1/8 + 1/16 │   36 │
                        │ eighth note  │    1/8     │   24 │
                        │ dotted 16th  │1/16 + 1/32 │   18 │
                        │  16th note   │    1/16    │   12 │
                        │ dotted 32nd  │1/32 + 1/64 │    9 │
                        │  32nd note   │    1/32    │    6 │
                        └───────┴──────┴───┘
